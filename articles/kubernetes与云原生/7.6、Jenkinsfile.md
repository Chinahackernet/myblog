pipeline {
    agent any

    stages {
        stage('Initialize') {
            steps {
                script {
                    // 定义用于存储解析后的包名和服务名的列表
                    def packages = []
                    def services = []
                    def jobs = [:]
                    def refreshCdn = false

                    // 调用自定义函数获取构建信息，并填充 packages 和 services 列表
                    refreshCdn = getBuildInfo(packages, services)

                    // 遍历 services 列表，为每个服务创建对应的部署任务
                    services.eachWithIndex { serviceName, index ->
                        def packageName = packages[index]
                        jobs[packageName] = transformIntoStep(serviceName, packageName)
                    }
                    
                    // 设置环境变量，用于后续的 steps 使用
                    env.REFRESH_CDN = refreshCdn
                    env.JOBS = jobs
                }
            }
        }

        stage('Pull Package') {
            // 根据条件是否拉取部署包
            when {
                expression { params.IS_PULL_PACKAGE }
            }
            steps {
                // 调用外部任务拉取部署包
                build job: 'ma_cicd_pull', parameters: [
                    string(name: 'Version', value: params.Version),
                    text(name: 'package_para', value: params.Packages)
                ]
            }
        }

        stage('Deploy Services') {
            // 并行部署所有服务
            steps {
                script {
                    parallel env.JOBS
                }
            }
        }
    }
}

// 自定义函数：解析构建信息并填充 packages 和 services 列表
def getBuildInfo(packages, services) {
    def refreshCdn = false
    def packageList = params.Packages.split('\n')
    for (package in packageList) {
        def appName = package.split('-')[0].trim()
        def serviceName = appName.contains("front") ? "ma-smartxmastatic.51ima.lo" : "ma-${appName}"
        def packageName = "${appName}.zip"
        packages.add(packageName)
        services.add(serviceName)
    }
    return refreshCdn
}

// 自定义函数：将服务名和包名转换为部署步骤
def transformIntoStep(serviceName, packageName) {
    return {
        node("master") {
            stage("Deploy: ${serviceName}") {
                def packageLocation = "${env.FTP_ADDR}ma/${params.Version}/${packageName}"
                echo "Package Location: ${packageLocation}"
                // 调用指定服务的部署任务
                build job: serviceName, parameters: [
                    string(name: 'FTPath', value: packageLocation)
                ]
            }
        }
    }
}
